```sql
CREATE VIEW contacted_registry AS
SELECT p.id, p.apollo_person_id, p.apollo_contact_id, p.company_domain,
  CASE WHEN p.company_domain IS NULL THEN NULL
       WHEN p.company_domain IN (SELECT domain FROM free_email_domains) THEN NULL
       ELSE p.company_domain END AS dedupable_domain,  -- NULL = don't dedup on domain
  p.status
FROM prospect p WHERE p.apollo_contact_id IS NOT NULL;
```
